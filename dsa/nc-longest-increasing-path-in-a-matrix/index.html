Run it in Java

Left is the snippet. Right is the output — one System.out.println at a time.

memo DFS on grid

int[][] matrix = {{9, 9, 4}, {6, 6, 8}, {2, 1, 1}};
int rows = 3, cols = 3;
int[][] memo = new int[rows][cols];
int dfs(int r, int c) {
  if (memo[r][c] != 0) return memo[r][c];
  int best = 1;
  for (int[] d : new int[][]{{1,0},{-1,0},{0,1},{0,-1}}) {
    int nr = r + d[0], nc = c + d[1];
    if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
    if (matrix[nr][nc] > matrix[r][c]) best = Math.max(best, 1 + dfs(nr, nc));
  }
  return memo[r][c] = best;
}
int ans = 0;
for (int r = 0; r < rows; r++)
  for (int c = 0; c < cols; c++) {
    ans = Math.max(ans, dfs(r, c));
    System.out.println("(" + r + "," + c + ")=" + memo[r][c]);
  }
System.out.println("lip=" + ans);
